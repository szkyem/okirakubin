onAuthStateChanged(auth, async (user) => { if (!user) { showLoginUI(); return; } try { const adminSnapshot = await get(ref(database, `admins/${user.uid}`)); const memberSnapshot = await get(ref(database, `members/${user.uid}`)); let role = null; if (adminSnapshot.exists() && adminSnapshot.val() === true) { role = "admin"; } else if (memberSnapshot.exists() && memberSnapshot.val() === true) { role = "member"; } if (!role) { await signOut(auth); showError("このアカウントは利用登録されていません。管理者に確認してください。"); return; } await set(ref(database, `loginUsers/${user.uid}`), { uid: user.uid, name: user.displayName || "", email: user.email || "", role, lastLoginAt: serverTimestamp() }); if (role === "admin") { location.href = "admin.html"; } else { location.href = "member.html"; } } catch (error) { console.error(error); showError("ログイン後の権限確認に失敗しました。"); } });
